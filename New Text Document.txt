document.addEventListener("DOMContentLoaded", () => {
  const containerEl = document.getElementById("konva-container");
  if (!containerEl) return;

  // Web Audio Context
  const AudioContext = window.AudioContext || window.webkitAudioContext;
  let audioCtx = null;

  function svirajMelodiju() {
    if (!audioCtx) audioCtx = new AudioContext();
    if (audioCtx.state === 'suspended') audioCtx.resume();

    const tonovi = [261.63, 329.63, 392.00];
    tonovi.forEach((f, i) => {
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.frequency.setValueAtTime(f, audioCtx.currentTime + i * 0.2);
      gain.gain.setValueAtTime(0.2, audioCtx.currentTime + i * 0.2);
      gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + i * 0.2 + 0.3);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start(audioCtx.currentTime + i * 0.2);
      osc.stop(audioCtx.currentTime + i * 0.2 + 0.3);
    });
  }

  // Konva Setup
  const stage = new Konva.Stage({
    container: "konva-container",
    width: containerEl.clientWidth,
    height: containerEl.clientHeight
  });

  const slojOsnova = new Konva.Layer();
  const slojHotspots = new Konva.Layer();
  const slojElementi = new Konva.Layer();

  // STANJE APLIKACIJE
  let trenutnaBaza = "meda";

      slojOsnova.add(new Konva.Circle({ x: cX - 55, y: cY - 105, radius: 25, fill: "#d69e2e", stroke: "#5c330a", strokeWidth: 3 }));

      slojOsnova.add(new Konva.Circle({ x: cX, y: cY - 60, radius: 65, fill: "#d69e2e", stroke: "#5c330a", strokeWidth: 4 }));
      
      // Lice
      slojOsnova.add(new Konva.Ellipse({ x: cX, y: cY - 48, radiusX: 22, radiusY: 16, fill: "#feebc8", stroke: "#b7791f", strokeWidth: 2 }));
      slojOsnova.add(new Konva.Ellipse({ x: cX, y: cY - 55, radiusX: 10, radiusY: 7, fill: "#322659" }));
      slojOsnova.add(new Konva.Circle({ x: cX - 22, y: cY - 72, radius: 6, fill: "#000" }));
      slojOsnova.add(new Konva.Circle({ x: cX + 22, y: cY - 72, radius: 6, fill: "#000" }));
    }

    slojOsnova.batchDraw();

    // Crtanje Hotspota i Ugrađenih elemenata
    if (bazaPodaci[trenutnaBaza]) {
      bazaPodaci[trenutnaBaza].zone.forEach(zona => {
        const apsX = cX + zona.x;
        const apsY = cY + zona.y;

        if (ugradjeniElementi[zona.id]) {
          // Ako ima element, crtaj element umesto Hotspota
          crtajElementNaZoni(zona.id, ugradjeniElementi[zona.id], apsX, apsY);
        } else {
          // Ako nema elementa, crtaj Hotspot
          const hotspot = new Konva.Circle({
            x: apsX, y: apsY,
            radius: 18,
            fill: "rgba(49, 130, 206, 0.4)",
            stroke: "#3182ce",
            strokeWidth: 2,
            cursor: "pointer"
          });

          // Pulsirajuća animacija
          const anim = new Konva.Tween({
            node: hotspot,
            duration: 0.8,
            radius: 22,
            opacity: 0.5,
            yoyo: true
          });
          anim.play();

          hotspot.on("click tap", () => otvoriPanelZaZonu(zona));
          slojHotspots.add(hotspot);
        }
      });
    }

    slojHotspots.batchDraw();
    slojElementi.batchDraw();
  }

  // ==================== LOGIKA ELEMENATA ====================
  function crtajElementNaZoni(zonaId, tipElementa, x, y) {
    const grupa = new Konva.Group({ x: x, y: y, cursor: 'pointer' });

    if (tipElementa === "zvucnik") {
      grupa.add(new Konva.Circle({ radius: 30, fill: '#805ad5', stroke: '#44337a', strokeWidth: 3 }));
      grupa.add(new Konva.Text({ text: '🎵', fontSize: 24, x: -12, y: -12 }));
      grupa.on("click tap", () => {
        svirajMelodiju();
        grupa.to({ scaleX: 1.2, scaleY: 1.2, duration: 0.1, yoyo: true });
        otvoriPodesavanja(zonaId, tipElementa);
      });
    } else if (tipElementa === "tekstura") {
      grupa.add(new Konva.Rect({ x: -35, y: -25, width: 70, height: 50, fill: '#dd6b20', cornerRadius: 8, stroke: '#7b341e', strokeWidth: 3 }));
      grupa.on("click tap", () => otvoriPodesavanja(zonaId, tipElementa));
    } else if (tipElementa === "rucka") {
      grupa.add(new Konva.Rect({ x: -70, y: -15, width: 140, height: 30, fill: '#38a169', cornerRadius: 15, stroke: '#1c4532', strokeWidth: 3 }));
      grupa.on("click tap", () => otvoriPodesavanja(zonaId, tipElementa));
    }
    // (Dodati ostale crteže po potrebi)

    slojElementi.add(grupa);
  }

  // ==================== UI PANEL LOGIKA ====================
  
  function prikaziPanel(nivo) {
    document.getElementById("panel-level-1").style.display = nivo === 1 ? "block" : "none";
    document.getElementById("panel-level-2").style.display = nivo === 2 ? "block" : "none";
    document.getElementById("panel-level-3").style.display = nivo === 3 ? "block" : "none";

    document.getElementById("step-2").classList.toggle("active", nivo >= 2);
    document.getElementById("step-3").classList.toggle("active", nivo === 3);
  }

  function otvoriPanelZaZonu(zona) {
    trenutnaZona = zona;
    document.getElementById("trenutna-zona-naslov").innerText = `Zona: ${zona.naziv}`;
    
    const container = document.getElementById("elements-container");
    container.innerHTML = "";

    zona.dozvoljeno.forEach(tip => {
      const el = elementiUI[tip];
      const btn = document.createElement("button");
      btn.className = "btn-element";
      btn.innerHTML = `<span class="el-icon">${el.ikona}</span><span class="el-name">${el.naziv}</span>`;
      
      btn.onclick = () => {
        ugradjeniElementi[zona.id] = tip;
        crtajBazu();
        otvoriPodesavanja(zona.id, tip);
      };
      container.appendChild(btn);
    });

    prikaziPanel(2);
  }

  function otvoriPodesavanja(zonaId, tipElementa) {
    trenutnaZona = bazaPodaci[trenutnaBaza].zone.find(z => z.id === zonaId);
    document.getElementById("podesavanje-naslov").innerText = elementiUI[tipElementa].naziv;
    prikaziPanel(3);
  }

  // Dugmići za nazad
  document.getElementById("btn-back-to-1").onclick = () => prikaziPanel(1);
  document.getElementById("btn-back-to-2").onclick = () => otvoriPanelZaZonu(trenutnaZona);

  // Uklanjanje
  document.getElementById("btn-ukloni-element").onclick = () => {
    ugradjeniElementi[trenutnaZona.id] = null;
    crtajBazu();
    prikaziPanel(2);
  };

  // Promena Baze
  document.querySelectorAll(".btn-base").forEach(btn => {
    btn.onclick = () => {
      document.querySelectorAll(".btn-base").forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
      trenutnaBaza = btn.dataset.base;
      ugradjeniElementi = { stomak: null, sape: null, usi: null, dno: null }; // reset
      crtajBazu();
      prikaziPanel(1);
    };
  });

  // Globalni Reset
  document.getElementById("btn-reset").onclick = () => {
    ugradjeniElementi = { stomak: null, sape: null, usi: null, dno: null };
    crtajBazu();
    prikaziPanel(1);
  };

  // Preuzimanje
  document.getElementById("btn-save").onclick = () => {
    const dataURL = stage.toDataURL({ pixelRatio: 2 });
    const link = document.createElement("a");
    link.download = `medicinska-igracka.png`;
    link.href = dataURL;
    link.click();
  };

  // Inicijalizacija
  crtajBazu();

  window.addEventListener("resize", () => {
    stage.width(containerEl.clientWidth);
    stage.height(containerEl.clientHeight);
    crtajBazu();
  });
});
